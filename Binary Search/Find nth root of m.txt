class Solution {
    public int nthRoot(int n, int m) {
        // code here
        int low=1,high=m;
        
        if(m==0)
        return 0;
        
        while(low<=high){
            int mid=low+(high-low)/2;
            
            long value=power(mid,n,m);
            
            if(value==m){
                return mid;
            }else if(value<m){
                low=mid+1;
            }else{
                high=mid-1;
            }
        }
        return -1;
    }
    
    long power(int mid,int n,int m){
        long result=1;
        
        for(int i=0;i<n;i++){
            result*=mid;
            if(result>m)
            return result;
        }
        return result;
    }
}
