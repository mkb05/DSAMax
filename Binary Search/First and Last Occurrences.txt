class GFG {
    ArrayList<Integer> find(int arr[], int x) {
        // code here
        int n=arr.length;
        
        int low=0,high=n-1;
        
        ArrayList<Integer> list=new ArrayList<>();
        
      
        int min=-1,max=-1;
        while(low<=high){
            int mid=low+(high-low)/2;
            
            if(arr[mid]==x){
                min=mid;
               
                high=mid-1;
            }else if(arr[mid]<x){
                low=mid+1;
            }else{
                high=mid-1;
            }
        }
        
        low=0;high=n-1;
       
       while(low<=high){
           int mid=low+(high-low)/2;
           
           if(arr[mid]==x){
               max=mid;
               low=mid+1;
           } else if(arr[mid]<x){
               low=mid+1;
           }else{
               high=mid-1;
           }
       }
       
       list.add(min);
       list.add(max);
      
       
       return list;
    }
}
