class Solution {
    public int peakElement(int[] arr) {
        // code here
        int n=arr.length;
        
        int low=0,high=n-1;
        
        
        while(low<high){
           int mid=(high-low)/2;
           mid+=low;
            
            if(arr[mid]<arr[mid+1]){
                low=mid+1;
            }else{
                high=mid;
            }
        }
        return low;
    }
}
