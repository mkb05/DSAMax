class Solution {
    public int activitySelection(int[] start, int[] finish) {
        // code here
        int count=0;
        
        PriorityQueue<int[]> pq=new PriorityQueue<>((a,b)->a[1]-b[1]);
        
        int n=start.length;
        
        for(int i=0;i<n;i++){
            pq.offer(new int[]{start[i],finish[i]});
        }
        
        int lastfinish=-1;
        
        while(!pq.isEmpty()){
            int cur[]=pq.poll();
            
            if(cur[0] > lastfinish){
                count++;
                lastfinish=cur[1];
            }
        }
        return count;
    }
}
