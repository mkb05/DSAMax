class Solution {
    public ArrayList<Integer> kLargest(int[] arr, int k) {
        // Your code here
        PriorityQueue<Integer> pq=new PriorityQueue<>(Collections.reverseOrder());
        ArrayList<Integer> ans=new ArrayList<>();
        for(int i:arr){
            pq.add(i);
        }
        
        for(int i=0;i<k;i++){
            ans.add(pq.poll());
        }
        
        return ans;
    }
}
