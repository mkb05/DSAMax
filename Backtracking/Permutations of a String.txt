class Solution {
    public ArrayList<String> findPermutation(String s) {
        // Code here
        ArrayList<String> result=new ArrayList<>();
        char[] chars=s.toCharArray();
        boolean used[]=new boolean[chars.length];
        
        Arrays.sort(chars);
        backtrack(result,used,new StringBuilder(),chars);
        
        return result;
    }
    
    static void backtrack(ArrayList<String> result,boolean[] used,StringBuilder path,char[] chars){
        if(path.length()==chars.length){
            result.add(path.toString());
            return;
        }
        
        for(int i=0;i<chars.length;i++){
            if(used[i])
            continue;
            
            if(i>0 && chars[i]==chars[i-1] && !used[i-1])
            continue;
            
            used[i]=true;
            path.append(chars[i]);
            
            backtrack(result,used,path,chars);
            
            path.deleteCharAt(path.length()-1);
            used[i]=false;
            
        }
    }
}
