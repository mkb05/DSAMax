class Solution {
    public static ArrayList<Integer> findUnion(int a[], int b[]) {
        // code here
        int m=a.length;
        int n=b.length;
        
        int i=0,j=0;
        ArrayList<Integer> list=new ArrayList<>();
        
        while(i<m && j<n){
           
           if(i>0 && a[i]==a[i-1]){
               i++;
               continue;
           }
           
           if(j>0 && b[j]==b[j-1]){
               j++;
               continue;
           }
           
           if(a[i] <b[j]){
               list.add(a[i]);
               i++;
           }else if(a[i]>b[j]){
               list.add(b[j]);
               j++;
           }else{
               list.add(a[i]);
               i++;
               j++;
           }
        }
        
        while (i < m) {
            if (i == 0 || a[i] != a[i - 1]) {
                list.add(a[i]);
            }
            i++;
        }

        // Remaining elements of b[]
        while (j < n) {
            if (j == 0 || b[j] != b[j - 1]) {
                list.add(b[j]);
            }
            j++;
        }
        
        return list;
    }
}
